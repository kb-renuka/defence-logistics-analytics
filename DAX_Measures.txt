// DAX measures — Defence Logistics & Operations Analytics
// In Power BI Desktop: Modeling > New Measure, paste each one individually.
// Assumes the table loaded via Power_Query_ETL.pq is named "logistics".

Total Cost (INR) =
SUM ( logistics[Monthly_Cost_INR] )

Avg Operational Readiness =
AVERAGE ( logistics[Operational_Readiness] )

Delayed Delivery % =
DIVIDE (
    CALCULATE ( COUNTROWS ( logistics ), logistics[Delivery_Status] = "Delayed" ),
    COUNTROWS ( logistics )
)

Total Supply Requests =
SUM ( logistics[Supply_Requests] )

Avg Fuel Consumption (L) =
AVERAGE ( logistics[Fuel_Consumption_Litres] )

Equipment Under Maintenance % =
DIVIDE (
    CALCULATE (
        COUNTROWS ( logistics ),
        logistics[Maintenance_Status] IN { "Under Maintenance", "Awaiting Parts" }
    ),
    COUNTROWS ( logistics )
)

MoM Cost Change % =
VAR CurrentCost = [Total Cost (INR)]
VAR PreviousCost =
    CALCULATE (
        [Total Cost (INR)],
        DATEADD ( logistics[Delivery_Date], -1, MONTH )
    )
RETURN
    DIVIDE ( CurrentCost - PreviousCost, PreviousCost )

Cost Per Personnel =
DIVIDE ( [Total Cost (INR)], SUM ( logistics[Personnel_Count] ) )

Readiness Rank by Region =
RANKX (
    ALL ( logistics[Region] ),
    CALCULATE ( [Avg Operational Readiness] ),
    ,
    ASC
)

High Readiness Units =
CALCULATE (
    DISTINCTCOUNT ( logistics[Unit_ID] ),
    logistics[Readiness_Band] = "High"
)
